\newcommand{\segbar}[5]{%
  \begin{scope}[shift={(#1)}]
    \pgfmathsetmacro{\segx}{-#2/2}
    \foreach \n/\c/\l/\lc in {#5} {%
      \pgfmathsetmacro{\segw}{#2*\n/#4}
      \fill[\c, draw=white, line width=0.4pt]
           (\segx, -#3/2) rectangle (\segx+\segw, #3/2);
      \ifx\l\empty\else
        \node[font=\scriptsize, text=\lc, inner sep=0pt]
             at (\segx+\segw/2, 0) {\l};
      \fi
      \pgfmathsetmacro{\segnext}{\segx+\segw}
      \global\let\segx\segnext
    }
  \end{scope}}

\newcommand{\keptbar}[2]{%
  \begin{scope}[shift={(#1)}]
    \pgfmathsetmacro{\pr}{#2*\nDrop/\Nfeat}
    \draw[btHilite, dashed, dash pattern=on 2pt off 1.5pt, line width=0.5pt]
         (-#2/2, -0.21) rectangle (-#2/2+\pr, 0.21);
    \fill[btPale] (-#2/2+\pr, -0.21) rectangle (#2/2, 0.21);
    \node[font=\scriptsize, text=btInk, inner sep=0pt] at (\pr/2, 0)
         {\nKeep{} kept features};
  \end{scope}}
